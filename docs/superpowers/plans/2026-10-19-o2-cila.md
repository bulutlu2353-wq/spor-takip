```dart
/// [seen]'den sonraki vuruşlar, eskiden yeniye (O2 cila spec §6.1). [seen] null → boş (ilk açılış).
List<BattleHit> unseenHits(BattleState state, DateTime? seen) {
  if (seen == null) return const [];
  return [for (final hit in state.recent.reversed) if (hit.date.isAfter(seen)) hit];
}

/// Ana sayfa kartında oynatılacak vuruş özeti (O2 cila spec §6.2).
class HitReplay {
  const HitReplay({required this.damage, required this.crit, required this.fromHpLeft, this.defeated});

  final int damage;
  final bool crit;

  /// Animasyon başında kalan can oranı (0–1); yenilme varsa yeni canavar tam canla başlar.
  final double fromHpLeft;

  /// Yenilen son canavar; yoksa null.
  final int? defeated;
}

HitReplay hitReplay(BattleState state, List<BattleHit> hits) {
  final damage = hits.fold(0, (sum, h) => sum + h.damage);
  final defeated = hits.expand((h) => h.defeated).lastOrNull;
  final before = defeated == null ? math.max(0, state.damage - damage) : 0;
  return HitReplay(
    damage: damage,
    crit: hits.any((h) => h.critShare > 0),
    fromHpLeft: (state.hp - before) / state.hp,
    defeated: defeated,
  );
}
```

```dart
/// En son görülen vuruşun tarihi, cihazda (O2 cila spec §6.1); okunamazsa null.
Future<DateTime?> readSeenHit() async {
  try {
    final raw = (await SharedPreferences.getInstance()).getString(battleSeenHitKey);
    return raw == null ? null : DateTime.tryParse(raw);
  } catch (_) {
    return null;
  }
}

Future<void> writeSeenHit(DateTime date) async {
  try {
    await (await SharedPreferences.getInstance()).setString(battleSeenHitKey, date.toIso8601String());
  } catch (_) {}
}
```

```dart
/// Portre görünümü (O2 cila spec §2).
enum MonsterLook { normal, defeated, fogged, lowHp }

/// Aşama halkası ve yer tutucu renkleri: bronz, gümüş, mor, altın (O2 cila spec §4).
const _tierColors = [Color(0xFFCD7F32), Color(0xFFC0C7D0), Color(0xFF9C4DFF), Color(0xFFFFC83D)];
const _tierIcons = [Icons.pest_control, Icons.pets, Icons.bolt, Icons.local_fire_department];
const _stampRed = Color(0xFFE53935);

const _grayscale = ColorFilter.matrix([
  0.2126, 0.7152, 0.0722, 0, 0, //
  0.2126, 0.7152, 0.0722, 0, 0,
  0.2126, 0.7152, 0.0722, 0, 0,
  0, 0, 0, 1, 0,
]);

/// Canavarın daire portresi: kadraj, vinyet, aşama halkası ve [look]'a göre katman;
/// görsel yoksa aşama renginde simge ve adının baş harfi.
class MonsterAvatar extends StatefulWidget {
  const MonsterAvatar({
    super.key,
    required this.index,
    this.size = 64,
    this.look = MonsterLook.normal,
    this.stampScale = 1,
  });

  final int index;
  final double size;
  final MonsterLook look;

  /// Yenilme anında mührün "vurulma" ölçeği.
  final double stampScale;

  @override
  State<MonsterAvatar> createState() => _MonsterAvatarState();
}

class _MonsterAvatarState extends State<MonsterAvatar> with SingleTickerProviderStateMixin {
  late final _pulse = AnimationController(vsync: this, duration: const Duration(milliseconds: 1200));

  @override
  void didChangeDependencies() {
    super.didChangeDependencies();
    _syncPulse();
  }

  @override
  void didUpdateWidget(MonsterAvatar oldWidget) {
    super.didUpdateWidget(oldWidget);
    _syncPulse();
  }

  void _syncPulse() {
    final on = widget.look == MonsterLook.lowHp && !MediaQuery.disableAnimationsOf(context);
    if (on && !_pulse.isAnimating) {
      _pulse.repeat(reverse: true);
    } else if (!on && _pulse.isAnimating) {
      _pulse.stop();
    }
  }

  @override
  void dispose() {
    _pulse.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final size = widget.size;
    final look = widget.look;
    final monster = monsterAt(widget.index);
    final tier = monsterTier(widget.index);
    final tierColor = _tierColors[tier - 1];
    final ring = switch (look) {
      MonsterLook.defeated => Colors.grey,
      MonsterLook.lowHp => Theme.of(context).colorScheme.error,
      _ => tierColor,
    };
    final name = monster.nameKey.tr();
    final framed = size * monster.zoom;

    Widget portrait = OverflowBox(
      minWidth: framed,
      maxWidth: framed,
      minHeight: framed,
      maxHeight: framed,
      alignment: Alignment(monster.focus.$1, monster.focus.$2),
      child: Image.asset(
        monster.imageAsset,
        width: framed,
        height: framed,
        fit: BoxFit.cover,
        cacheWidth: (framed * MediaQuery.devicePixelRatioOf(context)).ceil(),
        errorBuilder: (context, error, stackTrace) => Stack(
          alignment: Alignment.center,
          children: [
            Icon(_tierIcons[tier - 1], size: size * 0.5, color: tierColor),
            Positioned(
              right: size * 0.12,
              bottom: size * 0.08,
              child: Text(
                name.isEmpty ? '' : name.characters.first,
                style: TextStyle(fontSize: size * 0.2, fontWeight: FontWeight.w900, color: tierColor),
              ),
            ),
          ],
        ),
      ),
    );
    if (look == MonsterLook.defeated) {
      portrait = Opacity(opacity: 0.6, child: ColorFiltered(colorFilter: _grayscale, child: portrait));
    } else if (look == MonsterLook.fogged) {
      portrait = ImageFiltered(
        imageFilter: ImageFilter.blur(sigmaX: size * 0.08, sigmaY: size * 0.08),
        child: portrait,
      );
    }

    final avatar = AnimatedBuilder(
      animation: _pulse,
      builder: (context, child) {
        final glow = switch (look) {
          MonsterLook.defeated => 0.0,
          MonsterLook.lowHp => size * 0.2 * _pulse.value,
          _ => size * 0.12,
        };
        return Container(
          key: Key('monster_avatar_${monster.id}'),
          width: size,
          height: size,
          decoration: BoxDecoration(
            shape: BoxShape.circle,
            color: tierColor.withValues(alpha: 0.18),
            border: Border.all(color: ring, width: math.max(2, size * 0.04)),
            boxShadow: [if (glow > 0) BoxShadow(color: ring.withValues(alpha: 0.6), blurRadius: glow)],
          ),
          child: child,
        );
      },
      child: ClipOval(
        child: Stack(
          fit: StackFit.expand,
          children: [
            portrait,
            const DecoratedBox(
              decoration: BoxDecoration(
                gradient: RadialGradient(colors: [Colors.transparent, Color(0xB3000000)], stops: [0.6, 1.0]),
              ),
            ),
            if (look == MonsterLook.fogged) ...[
              const ColoredBox(key: Key('monster_fog'), color: Color(0x8C000000)),
              Center(
                child: Text('?', style: TextStyle(fontSize: size * 0.4, fontWeight: FontWeight.w900, color: tierColor)),
              ),
            ],
            if (look == MonsterLook.defeated)
              Center(
                key: const Key('monster_stamp'),
                child: Transform.scale(
                  scale: widget.stampScale,
                  child: Transform.rotate(
                    angle: -20 * math.pi / 180,
                    child: size < 56
                        ? Icon(Icons.close, color: _stampRed, size: size * 0.6)
                        : Container(
                            width: size * 0.9,
                            padding: const EdgeInsets.symmetric(horizontal: 4, vertical: 2),
                            decoration: BoxDecoration(
                              border: Border.all(color: _stampRed, width: 2),
                              borderRadius: const BorderRadius.all(Radius.circular(4)),
                            ),
                            child: FittedBox(
                              child: Text(
                                'gamification.battle.stamp'.tr(),
                                style: const TextStyle(color: _stampRed, fontWeight: FontWeight.w900),
                              ),
                            ),
                          ),
                  ),
                ),
              ),
          ],
        ),
      ),
    );
    return look == MonsterLook.lowHp ? KeyedSubtree(key: const Key('monster_pulse'), child: avatar) : avatar;
  }
}
```

```dart
/// Ana sayfadaki aktif canavar kartı (O2a spec §7); yüklenirken, hatada ya da savaş yokken görünmez.
/// Görülmemiş vuruşlarda vuruş ya da yenilme animasyonu oynar (O2 cila spec §6).
class HomeBattleCard extends ConsumerStatefulWidget {
  const HomeBattleCard({super.key});

  @override
  ConsumerState<HomeBattleCard> createState() => _HomeBattleCardState();
}

/// Bir evrenin süresi (ms): vuruş animasyonu ve yenilme anının iki yarısı.
const _phaseMs = 1200.0;

class _HomeBattleCardState extends ConsumerState<HomeBattleCard> with TickerProviderStateMixin {
  late final _hit = AnimationController(vsync: this, duration: const Duration(milliseconds: 1200));
  late final _defeat = AnimationController(vsync: this, duration: const Duration(milliseconds: 2400));
  HitReplay? _replay;
  var _checkedOnce = false;
  DateTime? _checked;

  @override
  void initState() {
    super.initState();
    ref.listenManual(playerSummaryProvider, (_, next) {
      final battle = next.value?.battle;
      if (battle != null) _check(battle);
    }, fireImmediately: true);
  }

  @override
  void dispose() {
    _hit.dispose();
    _defeat.dispose();
    super.dispose();
  }

  /// İlk açılışta yalnız görüldü yazar; sonra görülmemiş vuruşları yazar ve oynatır.
  Future<void> _check(BattleState battle) async {
    final newest = battle.recent.firstOrNull?.date;
    if (_checkedOnce && newest == _checked) return;
    _checkedOnce = true;
    _checked = newest;
    final seen = await readSeenHit();
    if (!mounted) return;
    if (seen == null) {
      await writeSeenHit(newest ?? DateTime.fromMillisecondsSinceEpoch(0));
      return;
    }
    final hits = unseenHits(battle, seen);
    if (hits.isEmpty) return;
    await writeSeenHit(hits.last.date);
    if (!mounted || MediaQuery.disableAnimationsOf(context)) return;
    final replay = hitReplay(battle, hits);
    setState(() => _replay = replay);
    try {
      await (replay.defeated == null ? _hit : _defeat).forward(from: 0).orCancel;
    } on TickerCanceled {
      return;
    }
    if (mounted) setState(() => _replay = null);
  }

  void _open() {
    _hit.stop();
    _defeat.stop();
    if (_replay != null) setState(() => _replay = null);
    context.push('/home/battle');
  }

  @override
  Widget build(BuildContext context) {
    final battle = ref.watch(playerSummaryProvider).value?.battle;
    if (battle == null) return const SizedBox.shrink();
    final replay = _replay;
    return Card(
      margin: const EdgeInsets.only(bottom: 12),
      clipBehavior: Clip.antiAlias,
      child: InkWell(
        key: const Key('home_battle_card'),
        onTap: _open,
        child: Padding(
          padding: const EdgeInsets.all(16),
          child: switch (replay) {
            null => _BattleRow(battle: battle),
            HitReplay(defeated: null) => AnimatedBuilder(
                animation: _hit,
                builder: (context, _) => _BattleRow(battle: battle, replay: replay, t: _hit.value),
              ),
            HitReplay(defeated: final int index) => AnimatedBuilder(
                animation: _defeat,
                builder: (context, _) => _defeat.value < 0.5
                    ? _DefeatMoment(index: index, t: _defeat.value * 2)
                    : _BattleRow(battle: battle, replay: replay, t: (_defeat.value - 0.5) * 2),
              ),
          },
        ),
      ),
    );
  }
}

/// Sönümlü yatay sarsıntı; [s] 0–1.
Offset _shake(double s) => Offset(math.sin(s * math.pi * 8) * 6 * (1 - s), 0);

/// "−340" ya da "−340 · KRİTİK!".
String hitFloatText(HitReplay r) =>
    r.crit ? 'gamification.battle.hit_crit'.tr(namedArgs: {'n': '${r.damage}'}) : '−${r.damage}';

/// Kartın içeriği; [replay] varken [t] (0–1) anına göre animasyonlu.
class _BattleRow extends StatelessWidget {
  const _BattleRow({required this.battle, this.replay, this.t = 1});

  final BattleState battle;
  final HitReplay? replay;
  final double t;

  @override
  Widget build(BuildContext context) {
    final theme = Theme.of(context);
    final muted = theme.textTheme.bodySmall?.copyWith(color: theme.colorScheme.onSurfaceVariant);
    final replay = this.replay;
    final ms = t * _phaseMs;
    Widget avatar = MonsterAvatar(index: battle.index, size: 72, look: activeLook(battle));
    double? bar;
    if (replay != null) {
      bar = lerpDouble(
        replay.fromHpLeft,
        battle.hpLeft / battle.hp,
        Curves.easeOut.transform((ms / 800).clamp(0.0, 1.0)),
      );
      if (replay.defeated == null) {
        final s = (ms / 400).clamp(0.0, 1.0);
        avatar = Stack(
          clipBehavior: Clip.none,
          children: [
            Transform.translate(offset: _shake(s), child: avatar),
            Positioned.fill(
              child: IgnorePointer(
                child: DecoratedBox(
                  decoration: BoxDecoration(
                    shape: BoxShape.circle,
                    color: Colors.red.withValues(alpha: 0.5 * (1 - s)),
                  ),
                ),
              ),
            ),
            Positioned(
              left: -40,
              right: -40,
              top: 24 - 40 * t,
              child: Opacity(
                opacity: 1 - t,
                child: Text(
                  hitFloatText(replay),
                  key: const Key('battle_hit_float'),
                  textAlign: TextAlign.center,
                  style: TextStyle(color: theme.colorScheme.error, fontWeight: FontWeight.w900, fontSize: 16),
                ),
              ),
            ),
          ],
        );
      } else {
        final a = (ms / 400).clamp(0.0, 1.0);
        avatar = Opacity(opacity: a, child: Transform.scale(scale: 0.8 + 0.2 * a, child: avatar));
      }
    }
    return Row(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        avatar,
        const SizedBox(width: 16),
        Expanded(
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              Text(battle.monster.nameKey.tr(), style: theme.textTheme.titleMedium?.copyWith(fontWeight: FontWeight.w800)),
              Text(guardianText(battle), style: muted),
              const SizedBox(height: 8),
              BattleHpBar(battle: battle, value: bar),
              const SizedBox(height: 8),
              WeakMuscleChips(monster: battle.monster),
              const SizedBox(height: 8),
              Text(lastHitText(battle), style: theme.textTheme.bodySmall),
              if (battle.healing)
                Text('gamification.battle.healing'.tr(),
                    style: theme.textTheme.bodySmall?.copyWith(color: theme.colorScheme.error)),
            ],
          ),
        ),
      ],
    );
  }
}

/// Yenilme anı: canavar sarsılır, sonra gri olur ve mühür vurulur; [t] 0–1.
class _DefeatMoment extends StatelessWidget {
  const _DefeatMoment({required this.index, required this.t});

  final int index;
  final double t;

  @override
  Widget build(BuildContext context) {
    final theme = Theme.of(context);
    final ms = t * _phaseMs;
    final stamped = ms >= 400;
    return Row(
      key: const Key('battle_defeat_moment'),
      children: [
        Transform.translate(
          offset: stamped ? Offset.zero : _shake((ms / 400).clamp(0.0, 1.0)),
          child: MonsterAvatar(
            index: index,
            size: 72,
            look: stamped ? MonsterLook.defeated : MonsterLook.normal,
            stampScale: 1.6 - 0.6 * Curves.easeOut.transform(((ms - 400) / 300).clamp(0.0, 1.0)),
          ),
        ),
        const SizedBox(width: 16),
        Expanded(
          child: Text(
            'gamification.battle.defeated'.tr(namedArgs: {'name': monsterAt(index).nameKey.tr()}),
            style: theme.textTheme.titleMedium?.copyWith(fontWeight: FontWeight.w900, color: theme.colorScheme.error),
          ),
        ),
      ],
    );
  }
}
```
