## Global Constraints

- Migration yok; sunucu değişikliği yok.
- Can: `round(span × 1,2 / 10) × 10`; sonsuz boss seviye 100–109; index ≥ 15 sonsuz boss.
- Hasar: `round(xp × (1 + bonus) × (1 + 0,5 × pay))`; `bonus = (güç + dayanıklılık + istikrar) / 600`; kritik payı yalnız antrenmanda.
- İyileşme: son sayılan antrenman gününden sonraki 8. günden itibaren her gün `round(hp × 0,03)`; yenilen canavar geri gelmez.

const monsters = <Monster>[

  Monster('rat_king', ['lats', 'hamstrings']),
  Monster('skeleton_guard', ['shoulders', 'glutes']),

  Monster('stone_golem', ['middle back', 'adductors']),
  Monster('bridge_troll', ['traps', 'abductors']),
  Monster('minotaur', ['lower back', 'forearms']),
  Monster('ash_wraith', ['chest', 'hamstrings']),
  Monster('basilisk', ['quadriceps', 'lats']),
  Monster('cyclops', ['shoulders', 'calves']),
  Monster('hydra', ['glutes', 'biceps']),
  Monster('lich', ['abdominals', 'middle back']),
  Monster('behemoth', ['triceps', 'adductors']),
  Monster('crimson_dragon', ['traps', 'lower back']),
  Monster('ancient_titan', ['forearms', 'abductors']),
];

const endlessIndex = 15;
const monsterHpFactor = 1.2;
const endlessFromLevel = 100;
const endlessToLevel = 110;

/// Index ≥ [endlessIndex] her zaman sonsuz boss.
Monster monsterAt(int index) => monsters[math.min(index, endlessIndex)];

/// Bekçi: rütbe aralığındaki XP × 1,2; sonsuz boss: seviye 100–109. 10'a yuvarlı.
int monsterHp(int index) {
  final (from, to) = index >= endlessIndex
      ? (endlessFromLevel, endlessToLevel)
      : (Rank.values[index].minLevel, Rank.values[index + 1].minLevel);
  var span = 0;
  for (var level = from; level < to; level++) {
    span += xpForNext(level);
  }
  return (span * monsterHpFactor / 10).round() * 10;
}

/// Bekçisi olduğu rütbe; sonsuz boss için null.
Rank? guardedRank(int index) => index >= endlessIndex ? null : Rank.values[index + 1];

/// Görsel aşama (1–4); sonsuz boss 4.
int monsterTier(int index) => guardedRank(index)?.tier ?? 4;

```dart
import '../../progress/domain/strength.dart';
import '../../progress/domain/weekly_summary.dart';
import 'fair_play.dart';
import 'streak.dart';

/// Stat kuralları (O2a spec §4).
const strengthWindowDays = 90;
const strengthTopExercises = 3;
const strengthFullRatio = 4.5;
const enduranceWindowDays = 28;
const enduranceSetsPerPoint = 3;
const consistencyWeeks = 8;
const minStat = 1;
const maxStat = 100;
const maxStatSum = 600;

class StatInputs {
  const StatInputs({required this.verdicts, required this.weeklyGoal, this.bodyWeightKg});

  /// `judgeSessions` çıktısı; yalnız sayılanlar kullanılır.
  final List<SessionVerdict> verdicts;
  final int weeklyGoal;

  /// Profil kilosu; null → Güç 1.
  final double? bodyWeightKg;
}

class BattleStats {
  const BattleStats({required this.strength, required this.endurance, required this.consistency});

  static const floor = BattleStats(strength: minStat, endurance: minStat, consistency: minStat);

  final int strength;
  final int endurance;
  final int consistency;

  /// Hasar bonusu, en çok 0,5.
  double get bonus => (strength + endurance + consistency) / maxStatSum;

  int get bonusPercent => (bonus * 100).round();
}

int _stat(num value) => value.round().clamp(minStat, maxStat);

/// [day]'in sonuna kadar biten sayılan oturumlarla statlar.
BattleStats statsOn(DateTime day, StatInputs inputs) {
  final today = localDay(day);
  final end = DateTime(today.year, today.month, today.day + 1);
  final counted = [
    for (final v in inputs.verdicts)
      if (v.counted && v.finishedAt.isBefore(end)) v,
  ];
  return BattleStats(
    strength: _strength(counted, today, inputs.bodyWeightKg),
    endurance: _endurance(counted, today),
    consistency: _consistency(counted, today, inputs.weeklyGoal),
  );
}

int _strength(List<SessionVerdict> counted, DateTime today, double? bodyWeightKg) {
  if (bodyWeightKg == null || bodyWeightKg <= 0) return minStat;
  final from = DateTime(today.year, today.month, today.day - (strengthWindowDays - 1));
  final best = <String, double>{};
  for (final v in counted) {
    if (v.finishedAt.isBefore(from)) continue;
    for (final set in v.countedSets) {
      final estimate = estimateOneRepMax(weightKg: set.weightKg, reps: set.reps);
      if (estimate != null && estimate > (best[set.exerciseId] ?? 0)) best[set.exerciseId] = estimate;
    }
  }
  if (best.isEmpty) return minStat;
  final top = (best.values.toList()..sort((a, b) => b.compareTo(a))).take(strengthTopExercises);
  final ratio = top.fold(0.0, (sum, e) => sum + e) / bodyWeightKg;
  return _stat(ratio / strengthFullRatio * 100);
}

int _endurance(List<SessionVerdict> counted, DateTime today) {
  final from = DateTime(today.year, today.month, today.day - (enduranceWindowDays - 1));
  var sets = 0;
  for (final v in counted) {
    if (!v.finishedAt.isBefore(from)) sets += v.countedSets.length;
  }
  return _stat(sets / enduranceSetsPerPoint);
}

int _consistency(List<SessionVerdict> counted, DateTime today, int weeklyGoal) {
  final thisWeek = startOfWeek(today);
  final from = DateTime(thisWeek.year, thisWeek.month, thisWeek.day - 7 * consistencyWeeks);
  final days = <DateTime, Set<DateTime>>{};
  for (final v in counted) {
    final day = localDay(v.finishedAt);
    if (day.isBefore(from) || !day.isBefore(thisWeek)) continue;
    days.putIfAbsent(startOfWeek(day), () => {}).add(day);
  }
  final hits = days.values.where((d) => d.length >= weeklyGoal).length;
  return _stat(hits * maxStat / consistencyWeeks);
}
```

Map<String, int> _muscleSets(List<SessionSet> sets, Map<String, Exercise> exercisesById) {
  final counts = <String, int>{};
  for (final set in sets) {
    final muscle = exercisesById[set.exerciseId]?.primaryMuscles.firstOrNull;
    if (muscle != null) counts[muscle] = (counts[muscle] ?? 0) + 1;
  }
  return counts;
}

/// Savaş kuralları (O2a spec §5).
const healGraceDays = 7;
const healPercent = 3;
const critExtra = 0.5;
const recentHitCount = 10;

class BattleHit {
  const BattleHit({
    required this.date,
    required this.source,
    required this.xp,
    required this.damage,
    required this.critShare,
    required this.monsterIndex,
    this.defeated = const [],
  });

  final DateTime date;
  final XpSource source;
  final int xp;
  final int damage;

  /// 0–1: zayıf kasa düşen sayılan set payı.
  final double critShare;

  /// Vuruşun başladığı canavar.
  final int monsterIndex;

  /// Bu vuruşla yenilen canavarlar.
  final List<int> defeated;
}

class BattleState {
  const BattleState({
    required this.index,
    required this.damage,
    required this.stats,
    this.recent = const [],
    this.healed = 0,
    this.healing = false,
  });

  /// Aktif canavar; ≥ [endlessIndex] sonsuz boss.
  final int index;

  /// Aktif canavara verilen hasar.
  final int damage;

  /// Bugünkü statlar.
  final BattleStats stats;

  /// Son vuruşlar, yeniden eskiye.
  final List<BattleHit> recent;

  /// Molalarda toplam iyileşen can.
  final int healed;

  /// Bugün iyileşme sürüyor.
  final bool healing;

  int get guardiansBeaten => math.min(index, endlessIndex);
  int get bossKills => math.max(0, index - endlessIndex);
  Monster get monster => monsterAt(index);
  int get hp => monsterHp(index);
  int get hpLeft => hp - damage;
}

DateTime _addDays(DateTime day, int days) => DateTime(day.year, day.month, day.day + days);

/// [events] eskiden yeniye sıralı (`compareXpEvents`), meydan okuma ödülleri dahil.
BattleState battle(List<XpEvent> events, StatInputs inputs, {required DateTime today}) {
  var index = 0;
  var damage = 0;
  var healed = 0;
  DateTime? lastWorkoutDay;
  DateTime? healedThrough;
  final statsByDay = <DateTime, BattleStats>{};
  final hits = <BattleHit>[];

  /// [lastWorkoutDay] + 8'den [day]'e (dahil) işlenmemiş her gün iyileşme.
  void healThrough(DateTime day) {
    final last = lastWorkoutDay;
    if (last == null) return;
    final first = _addDays(last, healGraceDays + 1);
    final previous = healedThrough;
    for (var d = previous == null || previous.isBefore(first) ? first : _addDays(previous, 1);
        !d.isAfter(day);
        d = _addDays(d, 1)) {
      final amount = math.min(damage, (monsterHp(index) * healPercent / 100).round());
      damage -= amount;
      healed += amount;
      healedThrough = d;
    }
  }

  for (final e in events) {
    final day = localDay(e.date);
    final workout = e.source == XpSource.workout && e.xp > 0;
    healThrough(workout ? _addDays(day, -1) : day);
    if (workout) lastWorkoutDay = day;
    if (e.xp <= 0) continue;
    final bonus = statsByDay.putIfAbsent(day, () => statsOn(day, inputs)).bonus;
    final weak = monsterAt(index).weakMuscles;
    final onWeak = e.muscleSets.entries.where((m) => weak.contains(m.key)).fold(0, (sum, m) => sum + m.value);
    final share = workout && e.sets > 0 ? math.min(1.0, onWeak / e.sets) : 0.0;
    final hit = (e.xp * (1 + bonus) * (1 + critExtra * share)).round();
    final start = index;
    final defeated = <int>[];
    damage += hit;
    while (damage >= monsterHp(index)) {
      damage -= monsterHp(index);
      defeated.add(index);
      index++;
    }
    hits.add(BattleHit(
      date: e.date,
      source: e.source,
      xp: e.xp,
      damage: hit,
      critShare: share,
      monsterIndex: start,
      defeated: defeated,
    ));
  }

  final now = localDay(today);
  healThrough(now);
  final last = lastWorkoutDay;
  return BattleState(
    index: index,
    damage: damage,
    stats: statsByDay[now] ?? statsOn(now, inputs),
    recent: hits.reversed.take(recentHitCount).toList(),
    healed: healed,
    healing: last != null && !now.isBefore(_addDays(last, healGraceDays + 1)),
  );
}

/// Rütbe için hem seviye eşiği hem bekçi gerekir (O2a spec §5.6); savaş yoksa yalnız seviye.
Rank gatedRank(int level, BattleState? battle) {
  final byLevel = rankFor(level);
  if (battle == null) return byLevel;
  return Rank.values[math.min(byLevel.index, battle.guardiansBeaten)];
}

class _Hits extends StatelessWidget {
  const _Hits({required this.hits});

        XpSource.workout => Icons.fitness_center,
        XpSource.record => Icons.emoji_events_outlined,
        XpSource.mealDay => Icons.restaurant,
        XpSource.challenge => Icons.flag_outlined,
        XpSource.quest => Icons.task_alt,

  Widget build(BuildContext context) {

    if (hits.isEmpty) return Text('gamification.battle.hits_empty'.tr(), style: theme.textTheme.bodyMedium);

        for (final (i, hit) in hits.indexed) ...[

            key: Key('battle_hit_$i'),
            contentPadding: EdgeInsets.zero,
            leading: Icon(_icon(hit.source)),
            title: Text('gamification.battle.hit_line'.tr(namedArgs: {'xp': '${hit.xp}', 'damage': '${hit.damage}'})),

          for (final index in hit.defeated)

              ),
            ),
        ],
      ],
    );
  }
}
